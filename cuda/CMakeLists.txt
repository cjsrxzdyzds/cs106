cmake_minimum_required(VERSION 3.18)
project(CudaOperatorLessons LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()
add_executable(cpu_reference_tests reference_tests.cpp)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(cpu_reference_tests PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME cpu_operator_references COMMAND cpu_reference_tests)
option(BUILD_CUDA_LESSONS "Build GPU tests when a CUDA compiler is available" ON)
if(BUILD_CUDA_LESSONS)
  include(CheckLanguage)
  check_language(CUDA)
  if(CMAKE_CUDA_COMPILER)
    enable_language(CUDA)
    add_executable(cuda_operator_tests operators.cu)
    set_target_properties(cuda_operator_tests PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
    foreach(operator vector_add reduce matmul softmax)
      add_test(NAME cuda_${operator} COMMAND cuda_operator_tests ${operator})
      set_tests_properties(cuda_${operator} PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 120)
    endforeach()
  else()
    message(STATUS "CUDA compiler not found: only CPU references are built; GPU tests are NOT validated")
  endif()
endif()
