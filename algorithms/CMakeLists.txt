cmake_minimum_required(VERSION 3.16)
project(InterviewAlgorithms LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ALGORITHMS_SANITIZE "Enable ASan and UBSan with GCC/Clang" OFF)
add_executable(algorithm_tests tests.cpp)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(algorithm_tests PRIVATE -Wall -Wextra -Wpedantic)
  if(ALGORITHMS_SANITIZE)
    target_compile_options(algorithm_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(algorithm_tests PRIVATE -fsanitize=address,undefined)
  endif()
endif()
enable_testing()
foreach(chapter RANGE 1 6)
  add_test(NAME algorithms_${chapter} COMMAND algorithm_tests ${chapter})
  set_tests_properties(algorithms_${chapter} PROPERTIES TIMEOUT 30)
endforeach()
